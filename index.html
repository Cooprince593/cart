<script>
lucide.createIcons();

(function () {
const bgMusic = document.getElementById('bg-music');
const bgVideo = document.getElementById('bg-video');
const musicBtn = document.getElementById('music-toggle');
const volumeSlider = document.getElementById('volume-slider');
const iconMuted = document.getElementById('icon-muted');
const iconPlaying = document.getElementById('icon-playing');
const iconPlay = document.getElementById('icon-play');
const iconPause = document.getElementById('icon-pause');
const welcomeOverlay = document.getElementById('welcome-overlay');

let currentPlaylist = [];
let currentTrackIndex = 0;

fetch('https://api.github.com/repos/Cooprince593/cart/contents/music')
.then(response => response.json())
.then(data => {
if(Array.isArray(data)) {
currentPlaylist = data.filter(file => file.name.endsWith('.mp3'));
if (currentPlaylist.length > 0) {
currentTrackIndex = Math.floor(Math.random() * currentPlaylist.length);
bgMusic.src = currentPlaylist[currentTrackIndex].download_url;
bgMusic.load();
}
}
})
.catch(err => console.error("Error conectando con la carpeta de música:", err));

bgMusic.volume = volumeSlider.value;

function updateIcons() {
if (!bgMusic.paused && bgMusic.volume > 0) {
if(iconMuted) iconMuted.style.display = 'none';
if(iconPlaying) iconPlaying.style.display = 'block';
} else {
if(iconMuted) iconMuted.style.display = 'block';
if(iconPlaying) iconPlaying.style.display = 'none';
}
}

function refreshMusicButton() {
if (bgMusic.paused) {
if(iconPlay) iconPlay.style.display = 'block';
if(iconPause) iconPause.style.display = 'none';
} else {
if(iconPlay) iconPlay.style.display = 'none';
if(iconPause) iconPause.style.display = 'block';
}
}

function playSpecificTrack(index) {
if (currentPlaylist.length === 0) return;
currentTrackIndex = index;
bgMusic.src = currentPlaylist[currentTrackIndex].download_url;
bgMusic.currentTime = 0;
bgMusic.load();
const playPromise = bgMusic.play();
if (playPromise !== undefined) {
playPromise.catch(err => console.log("La reproducción falló: ", err));
}
}

function playNextTrack() {
if (currentPlaylist.length === 0) return;
const nextIndex = (currentTrackIndex + 1) % currentPlaylist.length;
playSpecificTrack(nextIndex);
}

function playPrevTrack() {
if (currentPlaylist.length === 0) return;
const prevIndex = (currentTrackIndex - 1 + currentPlaylist.length) % currentPlaylist.length;
playSpecificTrack(prevIndex);
}

bgMusic.addEventListener('play', () => { updateIcons(); refreshMusicButton(); });
bgMusic.addEventListener('pause', () => { updateIcons(); refreshMusicButton(); });
bgMusic.addEventListener('volumechange', updateIcons);
bgMusic.addEventListener('error', () => {
console.warn("Se detectó un error en el audio. Pasando a la siguiente pista automáticamente...");
setTimeout(() => playNextTrack(), 500);
});

welcomeOverlay.addEventListener('click', () => {
document.body.classList.add('is-blurred');
if (bgVideo) {
bgVideo.play().catch(err => console.log("Video no pudo reproducirse automáticamente:", err));
}

const playPromise = bgMusic.play();
if (playPromise !== undefined) {
playPromise.catch(err => console.log("Música bloqueada:", err));
}
welcomeOverlay.classList.add('fade-out');
setTimeout(() => {
document.getElementById('center').classList.add('card-visible');
}, 650);
});

musicBtn.addEventListener('click', (e) => {
e.stopPropagation();
if (!bgMusic.paused) bgMusic.pause();
else bgMusic.play().catch(err => console.log(err));
});

volumeSlider.addEventListener('input', (e) => {
bgMusic.volume = parseFloat(e.target.value);
});

const trackProgress = document.getElementById('track-progress');
const musicPlayPause = document.getElementById('music-playpause');
const musicNext = document.getElementById('music-next');
const musicPrev = document.getElementById('music-prev');

if (bgMusic) {
bgMusic.addEventListener('timeupdate', () => {
if (!bgMusic.duration) return;
trackProgress.value = Math.round((bgMusic.currentTime / bgMusic.duration) * 100);
});
bgMusic.addEventListener('ended', () => {
playNextTrack();
});
}

if (trackProgress) {
trackProgress.addEventListener('input', (e) => {
if (!bgMusic.duration) return;
bgMusic.currentTime = (parseFloat(e.target.value) / 100) * bgMusic.duration;
});
}

if (musicPlayPause) {
musicPlayPause.addEventListener('click', () => {
if (bgMusic.paused) bgMusic.play().catch(err => console.log(err));
else bgMusic.pause();
});
}

if (musicNext) {
musicNext.addEventListener('click', () => {
playNextTrack();
});
}

if (musicPrev) {
musicPrev.addEventListener('click', () => {
if (bgMusic.currentTime > 3) {
bgMusic.currentTime = 0;
bgMusic.play().catch(e => console.log(e));
} else {
playPrevTrack();
}
});
}

})();

(function () {
const wrap = document.getElementById('card-wrap');
const card = document.getElementById('card');
const MAX = 12;
let raf;
let lastTime = 0;

wrap.addEventListener('mousemove', e => {
const now = performance.now();
if (now - lastTime < 16) return;
lastTime = now;

const rect = wrap.getBoundingClientRect();
const rx = (-(e.clientY - (rect.top + rect.height / 2)) / (rect.height / 2)) * MAX;
const ry = ((e.clientX - (rect.left + rect.width / 2)) / (rect.width / 2)) * MAX;
cancelAnimationFrame(raf);
raf = requestAnimationFrame(() => {
card.style.transform = `rotateX(${rx}deg) rotateY(${ry}deg) scale(1.03)`;
});
});

wrap.addEventListener('mouseleave', () => {
cancelAnimationFrame(raf);
card.style.transform = 'rotateX(0deg) rotateY(0deg) scale(1)';
});
})();
</script>
